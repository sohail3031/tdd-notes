import java.util.HashMap;

import java.util.Map;

  

import exceptions.*;

  

public class ShoppingCart {

    private Map<String, CartItem> items = new HashMap<>();

    private double percentageDiscount = 0.0;

  

    private static final int MAX_QUANTITY = 10;

    private static final double BULK_DISCOUNT_THRESHOLD = 150.0;

    private static final double BULK_DISCOUNT_PERCENTAGE = 0.10;

  

    private static final Map<String, Integer> VALID_DISCOUNT_CODES = new HashMap<>();

    static {

        VALID_DISCOUNT_CODES.put("HOLIDAY25", 25);

    }

  

    private String discountCode = null;

  

    public int getItemCount() {

        return items.values().stream().mapToInt(CartItem::getQuantity).sum();

    }

  

    public int getTotal() {

        int subtotal = items.values().stream().mapToInt(item -> item.getPrice() * item.getQuantity()).sum();

  

        if (subtotal > BULK_DISCOUNT_THRESHOLD) {

            subtotal -= (int) (subtotal * BULK_DISCOUNT_PERCENTAGE);

        }

  

        // Apply discount code if valid

        int discountValue = 0;

        if (discountCode != null && VALID_DISCOUNT_CODES.containsKey(discountCode)) {

            discountValue = VALID_DISCOUNT_CODES.get(discountCode);

        }

        if (discountValue > 0) {

            subtotal = (int) (subtotal * (1 - discountValue / 100.0));

        }

  

        // Apply percentage discount

        return (int) (subtotal * (1 - percentageDiscount / 100));

    }

  

    public void addItem(Product product, int quantity) {

        if (quantity > MAX_QUANTITY) {

            throw new InvalidCartOperationException("Maximum quantity exceeded");

        }

  

        items.put(product.getId(), new CartItem(product, quantity));

    }

  

    public void addItem(Product product) {

        addItem(product, 1);

    }

  

    public void removeItem(String id) {

        if (!items.containsKey(id)) {

            throw new ItemNotFoundException("Item not found");

        }

  

        CartItem item = items.get(id);

        if (item.getQuantity() > 1) {

            item.setQuantity(item.getQuantity() - 1);

        } else {

            items.remove(id);

        }

    }

  

    public void applyDiscount(double percentage) {

        this.percentageDiscount = percentage;

    }

  

    public void clear() {

        items.clear();

    }

  

    public void updateQuantity(String id, int quantity) {

        if (items.containsKey(id)) {

            CartItem item = items.get(id);

            item.setQuantity(quantity);

        }

    }

  

    public CartItem getItem(String id) {

        return items.get(id);

    }

  

    // TODO: Update this method to check against the list of valid codes and throw a InvalidDiscountCodeException for invalid codes.

    public void applyDiscountCode(String code) {

        this.discountCode = code;

    }

}
